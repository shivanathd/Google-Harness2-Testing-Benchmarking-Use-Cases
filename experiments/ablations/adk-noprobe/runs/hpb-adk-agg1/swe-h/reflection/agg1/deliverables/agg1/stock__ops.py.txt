def add(data, sku, qty):
    if qty <= 0:
        raise ValueError("quantity must be positive")
    data[sku] = data.get(sku, 0) + qty
    return data


def remove(data, sku, qty):
    current = data.get(sku, 0)
    if current - qty < 0:
        raise ValueError("quantity would go below zero")
    data[sku] = current - qty
    return data
